🪜 Step 2 — Replace Code (FULLY CORRECT)

"use client";

import { useState } from "react";

export default function Home() {
  const [message, setMessage] = useState("");

  return (
    <div style={{ padding: "20px" }}>

      <h1 style={{ color: "#8b5cf6" }}>AI Chat</h1>

      <input
        type="text"
        value={message}
        onChange={(e) => setMessage(e.target.value)}
        placeholder="Type your message..."
        style={{
          padding: "10px",
          width: "300px",
          marginRight: "10px"
        }}
      />

      <button
        onClick={() => alert(message)}
        style={{ padding: "10px" }}
      >
        Send
      </button>

    </div>
  );
}

🔍 Code Explanation

  • useState → Stores input value
  • value={message} → binds input
  • onChange → updates state
  • button click → shows message
